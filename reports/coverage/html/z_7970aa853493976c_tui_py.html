28 raw_path = self.query_one("#input_path", Input).value.strip()

44 def run_scan(self, path: str, uuid: str) -> None: 

45 """Runs the scan in a background worker.""" 

46 log = self.query_one("#log_area", RichLog) 

47 log.write(f"[yellow]Scanning:[/yellow] {path} ...") 

48 

49 def _do_work(): 

50 # NOTE: Do NOT update UI from here. Just do the heavy lifting. 

51 try: 

52 # Call the tool logic directly 

53 data = scan_directory(path) 

54 

55 # Output file is saved INSIDE the target directory 

56 output_filename = f"{uuid}.json" 

57 output_path = os.path.join(path, output_filename) 

58 

59 save_to_json(data, output_path) 

60 return output_path, len(data.get("files", [])), len(data.get("dirs", [])) 

61 except Exception as e: 

62 # Raise it so on_worker_state_changed catches it 

63 raise RuntimeError(f"Scan failed: {e}") 

64 

65 # Launch worker 

66 self.app.run_worker(_do_work, exclusive=True, thread=True)

68 def on_worker_state_changed(self, event: Worker.StateChanged) -> None: 

69 """Monitor worker state.""" 

70 if event.state == WorkerState.SUCCESS: 

71 output_path, n_files, n_dirs = event.worker.result 

72 self.log_message(f"[bold green]Success![/bold green] Found {n_files} files, {n_dirs} dirs.") 

73 self.log_message(f"Saved to: [underline]{output_path}[/underline]") 

74 elif event.state == WorkerState.ERROR: 

75 self.log_message(f"[bold red]Failed:[/bold red] {event.worker.error}")
